cmake_minimum_required(VERSION 3.29)
project(paradox VERSION 2.0.3 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT MSVC)
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-stdlib=libc++>)
    add_link_options(-stdlib=libc++)
endif()
include(FetchContent)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
FetchContent_Declare(endstone GIT_REPOSITORY https://github.com/EndstoneMC/endstone.git GIT_TAG b4aaa87d2c5305e7d340168f824ac2e2ea722e44)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_Declare(tomlplusplus GIT_REPOSITORY https://github.com/marzer/tomlplusplus.git GIT_TAG 30172438cee64926dc41fdd9c11fb3ba5b2ba9de)
FetchContent_Declare(httplib GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git GIT_TAG 3af7f2c16147f3fbc6e4d717032daf505dc1652c)
FetchContent_Declare(sqlite URL https://www.sqlite.org/2025/sqlite-amalgamation-3500400.zip URL_HASH SHA256=1d3049dd0f830a025a53105fc79fd2ab9431aea99e137809d064d8ee8356b032)
FetchContent_Declare(bedrock_protocol GIT_REPOSITORY https://github.com/EndstoneMC/bedrock-protocol.git GIT_TAG ec394c9a437386448f60f33a7610ca3fb90fa089
    PATCH_COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/native/tools/harden_protocol.py <SOURCE_DIR>)
set(HTTPLIB_REQUIRE_OPENSSL ON CACHE BOOL "" FORCE)
set(HTTPLIB_USE_ZLIB_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_BROTLI_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_ZSTD_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(BEDROCK_PROTOCOL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BEDROCK_PROTOCOL_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(endstone json tomlplusplus httplib sqlite bedrock_protocol)
add_library(paradox_sqlite STATIC ${sqlite_SOURCE_DIR}/sqlite3.c)
target_include_directories(paradox_sqlite PUBLIC ${sqlite_SOURCE_DIR})
target_compile_definitions(paradox_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DEFAULT_MEMSTATUS=0)
if(UNIX)
    target_link_libraries(paradox_sqlite PUBLIC ${CMAKE_DL_LIBS} pthread)
endif()
add_library(paradox_core STATIC native/src/engine.cpp native/src/store.cpp native/src/protocol.cpp)
target_include_directories(paradox_core PUBLIC native/include)
target_link_libraries(paradox_core PUBLIC nlohmann_json::nlohmann_json tomlplusplus::tomlplusplus paradox_sqlite bedrock::protocol)
endstone_add_plugin(paradox native/src/plugin.cpp native/src/web.cpp)
target_link_libraries(paradox PRIVATE paradox_core httplib::httplib)
target_compile_definitions(paradox PRIVATE PARADOX_VERSION="${PROJECT_VERSION}")
include(CTest)
if(BUILD_TESTING)
    add_executable(paradox_tests native/tests/tests.cpp)
    target_link_libraries(paradox_tests PRIVATE paradox_core)
    add_test(NAME paradox_native COMMAND paradox_tests)
    # Run manually against payloads captured from a disposable native server.
    add_executable(paradox_wire_check native/tests/wire_check.cpp)
    target_link_libraries(paradox_wire_check PRIVATE paradox_core)
endif()
install(TARGETS paradox LIBRARY DESTINATION . RUNTIME DESTINATION .)
install(FILES LICENSE native/references.lock.json DESTINATION .)
